// ---------- API 키: env var 우선, 없으면 .local/api-keys.json(gitignore 대상, 기기 로컬)
// 캐시, 그것도 없으면 딱 한 번 물어보고 캐시에 저장 (다음부턴 안 물어봄). 절대 vault의
// git 추적 파일(profile.md 등)에는 안 쓴다. ----------
async function readLocalKeys() {
  if (!(await app.vault.adapter.exists(LOCAL_KEYS_PATH))) return {};
  try {
    return JSON.parse(await app.vault.adapter.read(LOCAL_KEYS_PATH));
  } catch (e) {
    return {};
  }
}
async function writeLocalKeys(keys) {
  await app.vault.adapter.mkdir(".local");
  await app.vault.adapter.write(LOCAL_KEYS_PATH, JSON.stringify(keys, null, 2) + "\n");
}
async function getApiKey(envName, label, applyUrl) {
  if (process?.env?.[envName]) return process.env[envName];
  const cache = await readLocalKeys();
  if (cache[envName]) return cache[envName];
  const val = await tp.system.prompt(
    `${label} — 없으면 발급: ${applyUrl} (지금 없으면 비워두고 나중에 다시 실행해도 됨)`,
    ""
  );
  if (val) {
    cache[envName] = val;
    await writeLocalKeys(cache);
    new Notice(`${envName}를 .local/api-keys.json에 저장했습니다 (git 대상 아님, 다음부턴 안 물어봄).`, 6000);
  }
  return val || null;
}

function xmlItems(doc) {
  return Array.from(doc.getElementsByTagName("item"));
}
function xmlField(item, ...tagNames) {
  for (const tag of tagNames) {
    const el = item.getElementsByTagName(tag)[0];
    if (el && el.textContent.trim()) return el.textContent.trim();
  }
  return "";
}
// data.go.kr JSON 응답은 API마다 envelope 구조가 달라서, region_cd를 가진 배열을 재귀로 찾는다.
function findRowsWithKey(obj, key, depth = 0) {
  if (depth > 6 || obj == null) return null;
  if (Array.isArray(obj)) {
    if (obj.length && typeof obj[0] === "object" && obj[0] && key in obj[0]) return obj;
    for (const v of obj) {
      const found = findRowsWithKey(v, key, depth + 1);
      if (found) return found;
    }
    return null;
  }
  if (typeof obj === "object") {
    for (const v of Object.values(obj)) {
      const found = findRowsWithKey(v, key, depth + 1);
      if (found) return found;
    }
  }
  return null;
}

// "서울시 동대문구" 같은 흔한 축약형은 공식 지역명("서울특별시 동대문구")과 부분일치가
// 안 돼서 그대로 API에 쏘면 매칭이 안 된다. 여러 변형을 순서대로 시도한다:
// 1) 입력 그대로  2) 마지막 토큰만(보통 시/군/구, 전국적으로 대개 유일)
// 3) 첫 토큰(광역시/도로 추정) 제외  4) 흔한 광역시/도 축약어 → 공식명 치환
// ("광주"는 광주광역시/경기 광주시 둘 다 가능해 일부러 자동치환 목록에서 뺌 — 원본 그대로
// 검색해서 여러 개 뜨면 사용자가 고르게 둔다).
function regionQueryCandidates(input) {
  const trimmed = input.trim();
  const tokens = trimmed.split(/\s+/).filter(Boolean);
  const SIDO_ALIASES = {
    "서울": "서울특별시", "서울시": "서울특별시",
    "부산": "부산광역시", "부산시": "부산광역시",
    "대구": "대구광역시", "대구시": "대구광역시",
    "인천": "인천광역시", "인천시": "인천광역시",
    "대전": "대전광역시", "대전시": "대전광역시",
    "울산": "울산광역시", "울산시": "울산광역시",
    "세종": "세종특별자치시", "세종시": "세종특별자치시",
    "경기": "경기도",
    "강원": "강원특별자치도", "강원도": "강원특별자치도",
    "충북": "충청북도", "충남": "충청남도",
    "전북": "전북특별자치도", "전남": "전라남도",
    "경북": "경상북도", "경남": "경상남도",
    "제주": "제주특별자치도", "제주도": "제주특별자치도",
  };
  const candidates = [trimmed];
  if (tokens.length > 1) {
    candidates.push(tokens[tokens.length - 1]);
    candidates.push(tokens.slice(1).join(" "));
  }
  if (SIDO_ALIASES[tokens[0]]) {
    candidates.push([SIDO_ALIASES[tokens[0]], ...tokens.slice(1)].join(" "));
  }
  return [...new Set(candidates)].filter(Boolean);
}

async function lookupLawdCd(apiKey, regionQuery) {
  const url = `https://apis.data.go.kr/1741000/StanReginCd/getStanReginCdList?serviceKey=${encodeURIComponent(apiKey)}&pageNo=1&numOfRows=100&type=json&locatadd_nm=${encodeURIComponent(regionQuery)}&flag=Y`;
  const res = await fetch(url);
  const text = await res.text();
  await dumpDebug(`법정동코드_${regionQuery}`, text);
  const data = JSON.parse(text);
  const errHeader = data?.OpenAPI_ServiceResponse?.cmmMsgHeader;
  if (errHeader) {
    // 예: SERVICE_KEY_IS_NOT_REGISTERED_ERROR — 이 API 데이터셋에 활용신청 안 된 키.
    // 지역명 문제와 구분되게 바로 던져서 위(tryAutoLookup)에서 명확히 알려준다.
    throw new Error(
      `법정동코드 API 오류: ${errHeader.returnAuthMsg || errHeader.errMsg} (코드 ${errHeader.returnReasonCode}) — data.go.kr에서 "행정표준코드_법정동코드" API 활용신청 됐는지 확인하세요.`
    );
  }
  const rows = findRowsWithKey(data, "region_cd") || [];
  // locatadd_nm이 정확히 입력값을 포함하는 항목 우선, 5자리 코드로 집계
  const counts = new Map();
  for (const r of rows) {
    const code5 = String(r.region_cd ?? "").slice(0, 5);
    if (code5.length === 5) counts.set(code5, (counts.get(code5) || 0) + 1);
  }
  return [...counts.keys()];
}

async function lookupTransactions(apiKey, lawdCd, dealYmd) {
  // "Dev" 접미사 붙은 RTMSDataSvcAptTradeDev는 다른 데이터셋(15126468, 상세자료)이라
  // 이 프로젝트가 활용신청한 15126469(기본 자료)의 실제 오퍼레이션명인
  // RTMSDataSvcAptTrade(접미사 없음)와는 다른 서비스키 등록이 필요하다. 실제 승인된 키로
  // 실측(2026-08-07)해서 접미사 없는 쪽이 맞는 것과 아래 필드명 전부를 확인했다.
  const url = `https://apis.data.go.kr/1613000/RTMSDataSvcAptTrade/getRTMSDataSvcAptTrade?serviceKey=${encodeURIComponent(apiKey)}&LAWD_CD=${lawdCd}&DEAL_YMD=${dealYmd}&numOfRows=999&pageNo=1`;
  const res = await fetch(url);
  const text = await res.text();
  await dumpDebug(`실거래가_${lawdCd}_${dealYmd}`, text);
  const doc = new DOMParser().parseFromString(text, "text/xml");
  const errMsg = doc.getElementsByTagName("errMsg")[0]?.textContent;
  if (errMsg) {
    // 예: SERVICE_KEY_IS_NOT_REGISTERED_ERROR — 이 API 데이터셋에 활용신청 안 된 키.
    // 안 던지면 xmlItems(doc)이 그냥 빈 배열이라 "거래 없음"으로 오인해 달마다 헛수고로
    // 15번 반복하게 된다 — 바로 던져서 위(tryAutoLookup)에서 명확히 알려준다.
    const authMsg = doc.getElementsByTagName("returnAuthMsg")[0]?.textContent || errMsg;
    const code = doc.getElementsByTagName("returnReasonCode")[0]?.textContent || "";
    throw new Error(
      `아파트매매 실거래가 API 오류: ${authMsg} (코드 ${code}) — data.go.kr에서 "국토교통부_아파트 매매 실거래가 자료" API 활용신청 됐는지 확인하세요.`
    );
  }
  return xmlItems(doc).map((item) => ({
    아파트명: xmlField(item, "aptNm", "아파트"),
    거래금액만원: xmlField(item, "dealAmount", "거래금액"),
    전용면적: xmlField(item, "excluUseAr", "전용면적"),
    층: xmlField(item, "floor", "층"),
    년: xmlField(item, "dealYear", "년"),
    월: xmlField(item, "dealMonth", "월"),
    일: xmlField(item, "dealDay", "일"),
  }));
}

function yyyymmAdd(yyyymm, n) {
  const y = parseInt(yyyymm.slice(0, 4), 10);
  const m = parseInt(yyyymm.slice(4, 6), 10);
  const d = new Date(y, m - 1 + n, 1);
  return `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, "0")}`;
}

function currentYyyymm() {
  const d = new Date();
  return `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, "0")}`;
}

// 계약연월을 안 물어본다 — 최근 달부터 거꾸로 스캔해서 "가장 최근 신고된 거래"를 찾는다.
// (매수 시점 가격이 아니라 "지금 기준 최신 신고가"가 필요하다는 요청 반영)
async function tryAutoLookup(아파트명, 지역) {
  const apiKey = await getApiKey(
    "DATA_GO_KR_KEY",
    "공공데이터포털 인증키(디코딩 키)",
    "https://www.data.go.kr/data/15126469/openapi.do"
  );
  if (!apiKey) {
    new Notice("DATA_GO_KR_KEY가 없어 실거래가 자동조회를 건너뜁니다. 수동 입력으로 진행합니다.", 6000);
    return null;
  }
  try {
    const candidates = regionQueryCandidates(지역);
    let codes = [];
    for (const candidate of candidates) {
      codes = await lookupLawdCd(apiKey, candidate);
      if (codes.length > 0) break;
    }
    if (codes.length === 0) {
      new Notice(
        `"${지역}" — ${candidates.length}가지 변형(${candidates.join(", ")})으로 찾아봤지만 법정동코드를 못 찾았어요. 수동 입력으로 진행합니다.`,
        8000
      );
      return null;
    }
    let lawdCd = codes[0];
    if (codes.length > 1) {
      lawdCd = await tp.system.suggester(codes, codes, false, "여러 지역이 검색됨 — 정확한 지역코드 선택");
    }

    const thisMonth = currentYyyymm();
    const MONTHS_TO_SCAN = 15; // 최근 15개월 안에서 스캔 (그 아파트 거래가 뜸하면 더 늘려야 함)
    const inputName = norm(아파트명);
    let matches = [];
    let scannedYmd = null;
    const allRows = []; // 정확 매칭 실패 시 후보 선택지로 재활용
    for (let i = 0; i < MONTHS_TO_SCAN && matches.length === 0; i++) {
      const tryYmd = yyyymmAdd(thisMonth, -i);
      const rows = await lookupTransactions(apiKey, lawdCd, tryYmd);
      for (const r of rows) allRows.push({ ...r, _ymd: tryYmd });
      // 정부 등록명은 보통 "중앙하이츠빌"처럼 동 이름·"아파트" 접미사 없이 짧게만 온다.
      // 사용자는 "이문동 중앙하이츠빌 아파트"처럼 길게 입력하는 경우가 많아서, 등록명이
      // 사용자 입력의 부분문자열인지(반대 방향 X) 양쪽 다 확인한다.
      const found = rows.filter((r) => {
        const regName = norm(r.아파트명);
        return regName && (inputName.includes(regName) || regName.includes(inputName));
      });
      if (found.length > 0) {
        matches = found;
        scannedYmd = tryYmd;
      }
    }

    // 부분문자열로도 못 찾으면(오타·줄임말 등, 예: "중앙하이츠"↔실제 "중앙하이츠빌") 그냥
    // 포기하지 않고 스캔 기간에 실제 등록된 단지명 중에서 고르게 한다 — Templater suggester가
    // 자체적으로 fuzzy 검색을 지원해서 목록에서 타이핑하며 좁혀나갈 수 있다.
    if (matches.length === 0 && allRows.length > 0) {
      const uniqueNames = [...new Set(allRows.map((r) => r.아파트명))].sort();
      const picked = await tp.system.suggester(
        uniqueNames,
        uniqueNames,
        false,
        `"${아파트명}"과 정확히 일치하는 이름을 못 찾았어요 — 최근 ${MONTHS_TO_SCAN}개월 실제 거래 단지명 중에서 선택 (Esc: 수동 입력)`
      );
      if (picked) {
        matches = allRows.filter((r) => r.아파트명 === picked);
        scannedYmd = matches[0]?._ymd ?? null;
      }
    }

    if (matches.length === 0) {
      new Notice(`"${아파트명}" 최근 ${MONTHS_TO_SCAN}개월 내 거래를 못 찾았어요 (raw/_api-debug/ 원본 확인 가능). 수동 입력으로 진행합니다.`, 8000);
      return null;
    }

    let picked = matches[matches.length - 1];
    if (matches.length > 1) {
      const labels = matches.map((m) => `${m.아파트명} | ${m.거래금액만원}만원 | ${m.전용면적}㎡ | ${m.층}층 | ${m.년}.${m.월}.${m.일}`);
      const idx = await tp.system.suggester(labels, matches.map((_, i) => i), false, `${scannedYmd} 신고분 중 선택 (가장 최근 신고월)`);
      picked = matches[idx] ?? picked;
    }
    return picked;
  } catch (e) {
    // 무슨 단계에서 실패했는지(예: require("obsidian") 자체가 이 실행 컨텍스트에서 막혀
    // dumpDebug까지 못 갔을 가능성)를 남기려고, 어떤 debug 파일도 안 남았을 상황을 대비해
    // 에러 stack을 직접 저장한다. Notice는 몇 초 뒤 사라져서 메시지가 안 남기 때문.
    const errName = `온보딩_오류_${Date.now()}`;
    await dumpDebug(errName, `${아파트명} / ${지역}\n\n${e.stack ?? e.message ?? String(e)}`);
    new Notice(`실거래가 자동조회 중 오류: ${e.message} (raw/_api-debug/${errName}.txt에 자세히 남김). 수동 입력으로 진행합니다.`, 10000);
    return null;
  }
}

// 대개 잘 걸림) 새로 쌓지 않고 그 자리에서 덮어쓴다 — 예전엔 재시도할 때마다 계속
// push만 해서 같은 집이 여러 번(오타 시도까지 포함) 중복으로 쌓였다.
function upsertHousing(list, entry) {
  const key = norm(entry.아파트명);
  const idx = list.findIndex((h) => {
    const hKey = norm(h.아파트명);
    return hKey && key && (hKey.includes(key) || key.includes(hKey));
  });
  if (idx >= 0) list[idx] = entry;
  else list.push(entry);
}
