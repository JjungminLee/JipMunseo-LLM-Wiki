// ---------- API 키: env var 우선, 없으면 .local/api-keys.json(gitignore 대상, 기기 로컬)
// 캐시, 그것도 없으면 딱 한 번 물어보고 캐시에 저장 (다음부턴 안 물어봄). 절대 vault의
// git 추적 파일(profile.md 등)에는 안 쓴다. ----------
async function readLocalKeys() {
  if (!(await app.vault.adapter.exists(LOCAL_KEYS_PATH))) return {};
  try {
    return JSON.parse(await app.vault.adapter.read(LOCAL_KEYS_PATH));
  } catch (e) {
    return {};
  }
}
async function writeLocalKeys(keys) {
  await app.vault.adapter.mkdir(".local");
  await app.vault.adapter.write(LOCAL_KEYS_PATH, JSON.stringify(keys, null, 2) + "\n");
}
async function getApiKey(envName, label, applyUrl) {
  if (process?.env?.[envName]) return process.env[envName];
  const cache = await readLocalKeys();
  if (cache[envName]) return cache[envName];
  const val = await tp.system.prompt(
    `${label} — 없으면 발급: ${applyUrl} (지금 없으면 비워두고 나중에 다시 실행해도 됨)`,
    ""
  );
  if (val) {
    cache[envName] = val;
    await writeLocalKeys(cache);
    new Notice(`${envName}를 .local/api-keys.json에 저장했습니다 (git 대상 아님, 다음부턴 안 물어봄).`, 6000);
  }
  return val || null;
}

function xmlItems(doc) {
  return Array.from(doc.getElementsByTagName("item"));
}
function xmlField(item, ...tagNames) {
  for (const tag of tagNames) {
    const el = item.getElementsByTagName(tag)[0];
    if (el && el.textContent.trim()) return el.textContent.trim();
  }
  return "";
}
// data.go.kr JSON 응답은 API마다 envelope 구조가 달라서, region_cd를 가진 배열을 재귀로 찾는다.
function findRowsWithKey(obj, key, depth = 0) {
  if (depth > 6 || obj == null) return null;
  if (Array.isArray(obj)) {
    if (obj.length && typeof obj[0] === "object" && obj[0] && key in obj[0]) return obj;
    for (const v of obj) {
      const found = findRowsWithKey(v, key, depth + 1);
      if (found) return found;
    }
    return null;
  }
  if (typeof obj === "object") {
    for (const v of Object.values(obj)) {
      const found = findRowsWithKey(v, key, depth + 1);
      if (found) return found;
    }
  }
  return null;
}

async function dumpDebug(name, text) {
  try {
    await app.vault.adapter.mkdir("raw/_api-debug");
    await app.vault.adapter.write(`raw/_api-debug/${name}.txt`, text);
  } catch (e) { /* 디버그 저장 실패는 무시 — 조회 자체는 계속 진행 */ }
}

async function lookupLawdCd(apiKey, regionQuery) {
  const url = `https://apis.data.go.kr/1741000/StanReginCd/getStanReginCdList?serviceKey=${encodeURIComponent(apiKey)}&pageNo=1&numOfRows=100&type=json&locatadd_nm=${encodeURIComponent(regionQuery)}&flag=Y`;
  const { requestUrl } = require("obsidian");
  const res = await requestUrl({ url, throw: false });
  await dumpDebug(`법정동코드_${regionQuery}`, res.text ?? "");
  const data = JSON.parse(res.text);
  const rows = findRowsWithKey(data, "region_cd") || [];
  // locatadd_nm이 정확히 입력값을 포함하는 항목 우선, 5자리 코드로 집계
  const counts = new Map();
  for (const r of rows) {
    const code5 = String(r.region_cd ?? "").slice(0, 5);
    if (code5.length === 5) counts.set(code5, (counts.get(code5) || 0) + 1);
  }
  return [...counts.keys()];
}

async function lookupTransactions(apiKey, lawdCd, dealYmd) {
  const url = `https://apis.data.go.kr/1613000/RTMSDataSvcAptTradeDev/getRTMSDataSvcAptTradeDev?serviceKey=${encodeURIComponent(apiKey)}&LAWD_CD=${lawdCd}&DEAL_YMD=${dealYmd}&numOfRows=999&pageNo=1`;
  const { requestUrl } = require("obsidian");
  const res = await requestUrl({ url, throw: false });
  await dumpDebug(`실거래가_${lawdCd}_${dealYmd}`, res.text ?? "");
  const doc = new DOMParser().parseFromString(res.text, "text/xml");
  return xmlItems(doc).map((item) => ({
    아파트명: xmlField(item, "aptNm", "아파트"),
    거래금액만원: xmlField(item, "dealAmount", "거래금액"),
    전용면적: xmlField(item, "excluUseAr", "전용면적"),
    층: xmlField(item, "floor", "층"),
    년: xmlField(item, "dealYear", "년"),
    월: xmlField(item, "dealMonth", "월"),
    일: xmlField(item, "dealDay", "일"),
  }));
}

function yyyymmAdd(yyyymm, n) {
  const y = parseInt(yyyymm.slice(0, 4), 10);
  const m = parseInt(yyyymm.slice(4, 6), 10);
  const d = new Date(y, m - 1 + n, 1);
  return `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, "0")}`;
}

function currentYyyymm() {
  const d = new Date();
  return `${d.getFullYear()}${String(d.getMonth() + 1).padStart(2, "0")}`;
}

// 계약연월을 안 물어본다 — 최근 달부터 거꾸로 스캔해서 "가장 최근 신고된 거래"를 찾는다.
// (매수 시점 가격이 아니라 "지금 기준 최신 신고가"가 필요하다는 요청 반영)
async function tryAutoLookup(아파트명, 지역) {
  const apiKey = await getApiKey(
    "DATA_GO_KR_KEY",
    "공공데이터포털 인증키(디코딩 키)",
    "https://www.data.go.kr/data/15126469/openapi.do"
  );
  if (!apiKey) {
    new Notice("DATA_GO_KR_KEY가 없어 실거래가 자동조회를 건너뜁니다. 수동 입력으로 진행합니다.", 6000);
    return null;
  }
  try {
    const codes = await lookupLawdCd(apiKey, 지역);
    if (codes.length === 0) {
      new Notice(`"${지역}"에 해당하는 법정동코드를 못 찾았어요. 수동 입력으로 진행합니다.`, 6000);
      return null;
    }
    let lawdCd = codes[0];
    if (codes.length > 1) {
      lawdCd = await tp.system.suggester(codes, codes, false, "여러 지역이 검색됨 — 정확한 지역코드 선택");
    }

    const thisMonth = currentYyyymm();
    const MONTHS_TO_SCAN = 15; // 최근 15개월 안에서 스캔 (그 아파트 거래가 뜸하면 더 늘려야 함)
    let matches = [];
    let scannedYmd = null;
    for (let i = 0; i < MONTHS_TO_SCAN && matches.length === 0; i++) {
      const tryYmd = yyyymmAdd(thisMonth, -i);
      const rows = await lookupTransactions(apiKey, lawdCd, tryYmd);
      const found = rows.filter((r) => r.아파트명.replace(/\s/g, "").includes(아파트명.replace(/\s/g, "")));
      if (found.length > 0) {
        matches = found;
        scannedYmd = tryYmd;
      }
    }

    if (matches.length === 0) {
      new Notice(`"${아파트명}" 최근 ${MONTHS_TO_SCAN}개월 내 거래를 못 찾았어요 (raw/_api-debug/ 원본 확인 가능). 수동 입력으로 진행합니다.`, 8000);
      return null;
    }

    let picked = matches[matches.length - 1];
    if (matches.length > 1) {
      const labels = matches.map((m) => `${m.아파트명} | ${m.거래금액만원}만원 | ${m.전용면적}㎡ | ${m.층}층 | ${m.년}.${m.월}.${m.일}`);
      const idx = await tp.system.suggester(labels, matches.map((_, i) => i), false, `${scannedYmd} 신고분 중 선택 (가장 최근 신고월)`);
      picked = matches[idx] ?? picked;
    }
    return picked;
  } catch (e) {
    new Notice(`실거래가 자동조회 중 오류: ${e.message}. 수동 입력으로 진행합니다.`, 8000);
    return null;
  }
}
